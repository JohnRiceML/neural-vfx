# native/infer: route B ScreenDetail runner (HLSL pre/post + ONNX Runtime DirectML EP). OWNER: native inference engineer.
#
# Third-party binaries are fetched (pinned version + SHA-256) from nuget.org into <build>/_deps at configure time
# and copied next to nvfx.dll. nvfx.dll does NOT import them: detail.cpp loads DirectML.dll and onnxruntime.dll by
# full path from its own directory on the first nvfx_detail_create, so the solver keeps working without them and
# System32's older DirectML.dll / onnxruntime.dll are never picked up by accident.

set(NVFX_DEPS ${CMAKE_BINARY_DIR}/_deps)
set(NVFX_DML_VERSION 1.15.4)
set(NVFX_DML_SHA256 4E7CB7DDCE8CF837A7A75DC029209B520CA0101470FCDF275C1F49736A3615B9)
set(NVFX_ORT_VERSION 1.24.4)
set(NVFX_ORT_SHA256 57E9F11B73437BEF7A309496135D4C1F96B1A8E9DDBA60013FA27BFC1D788681)

# fetch <id> <version> <sha256> <dest> <patterns...>: download the .nupkg (a zip), extract only `patterns`
function(nvfx_fetch_nupkg id version sha dest)
  set(stamp ${dest}/.nvfx_${version})
  if(EXISTS ${stamp})
    return()
  endif()
  string(TOLOWER ${id} lid)
  set(pkg ${NVFX_DEPS}/dl/${lid}.${version}.nupkg)
  set(url https://api.nuget.org/v3-flatcontainer/${lid}/${version}/${lid}.${version}.nupkg)
  string(TOLOWER ${sha} sha_l)
  set(have "")
  if(EXISTS ${pkg})
    file(SHA256 ${pkg} have)
  endif()
  if(NOT have STREQUAL sha_l)
    message(STATUS "nvfx infer: fetching ${id} ${version}")
    file(MAKE_DIRECTORY ${NVFX_DEPS}/dl)
    file(DOWNLOAD ${url} ${pkg} STATUS st TLS_VERIFY ON)
    list(GET st 0 code)
    if(NOT code EQUAL 0)
      # Some CMake builds (e.g. the WinLibs/MinGW one on this machine) ship without a CA bundle and cannot verify
      # TLS: fall back to Windows' own curl.exe (Schannel, system certificate store). Integrity = the SHA-256 below.
      find_program(NVFX_CURL curl HINTS "$ENV{SystemRoot}/System32")
      if(NOT NVFX_CURL)
        message(FATAL_ERROR "download of ${url} failed (${st}) and curl.exe was not found")
      endif()
      message(STATUS "nvfx infer: CMake download failed (${st}), retrying with ${NVFX_CURL}")
      execute_process(COMMAND ${NVFX_CURL} -sSfL -o ${pkg} ${url} RESULT_VARIABLE rc)
      if(NOT rc EQUAL 0)
        message(FATAL_ERROR "download of ${url} failed: curl exit ${rc}")
      endif()
    endif()
    file(SHA256 ${pkg} have)
    string(TOLOWER "${have}" have)
    if(NOT have STREQUAL sha_l)
      file(REMOVE ${pkg})
      message(FATAL_ERROR "${pkg}: SHA-256 mismatch (got ${have}, expected ${sha_l})")
    endif()
  endif()
  file(REMOVE_RECURSE ${dest})
  file(ARCHIVE_EXTRACT INPUT ${pkg} DESTINATION ${dest} PATTERNS ${ARGN})
  file(TOUCH ${stamp})
endfunction()

set(NVFX_DML_DIR ${NVFX_DEPS}/directml-${NVFX_DML_VERSION})
set(NVFX_ORT_DIR ${NVFX_DEPS}/onnxruntime-dml-${NVFX_ORT_VERSION})
nvfx_fetch_nupkg(Microsoft.AI.DirectML ${NVFX_DML_VERSION} ${NVFX_DML_SHA256} ${NVFX_DML_DIR}
                 include/* bin/x64-win/DirectML.dll LICENSE*.txt ThirdPartyNotices.txt)
nvfx_fetch_nupkg(Microsoft.ML.OnnxRuntime.DirectML ${NVFX_ORT_VERSION} ${NVFX_ORT_SHA256} ${NVFX_ORT_DIR}
                 build/native/include/* runtimes/win-x64/native/onnxruntime.dll
                 runtimes/win-x64/native/onnxruntime_providers_shared.dll LICENSE* ThirdPartyNotices.txt)
set(NVFX_INFER_DLLS ${NVFX_DML_DIR}/bin/x64-win/DirectML.dll
                    ${NVFX_ORT_DIR}/runtimes/win-x64/native/onnxruntime.dll
                    ${NVFX_ORT_DIR}/runtimes/win-x64/native/onnxruntime_providers_shared.dll)
file(COPY ${NVFX_INFER_DLLS} DESTINATION ${CMAKE_BINARY_DIR}/bin)

# ---- HLSL: compiled with dxc at build time and embedded (same pattern as native/sim)
set(INFER_HLSL_DIR ${CMAKE_CURRENT_SOURCE_DIR}/shaders)
set(INFER_GEN_DIR ${CMAKE_CURRENT_BINARY_DIR}/gen)
file(MAKE_DIRECTORY ${INFER_GEN_DIR}/infer_shaders)
set(INFER_SHADERS detail:pre_s2 detail:pre_s1 detail:pre_dml_s2 detail:pre_dml_s1 detail:lowfreq detail:fin
                   detail:post_s2 detail:post_s1)
set(INFER_DXC_FLAGS -nologo -T cs_6_6 -HV 2021 -enable-16bit-types -O3 -Gis -WX)
set(infer_headers "")
set(infer_includes "")
set(infer_table "")
foreach(item ${INFER_SHADERS})
  string(REPLACE ":" ";" parts ${item})
  list(GET parts 0 file)
  list(GET parts 1 entry)
  set(out ${INFER_GEN_DIR}/infer_shaders/${entry}.h)
  add_custom_command(
    OUTPUT ${out}
    COMMAND ${NVFX_DXC} ${INFER_DXC_FLAGS} -E ${entry} -Vn g_detail_${entry} -Fh ${out} ${INFER_HLSL_DIR}/${file}.hlsl
    DEPENDS ${INFER_HLSL_DIR}/${file}.hlsl
    COMMENT "dxc ${file}.hlsl:${entry}"
    VERBATIM)
  list(APPEND infer_headers ${out})
  string(APPEND infer_includes "#include \"infer_shaders/${entry}.h\"\n")
  string(APPEND infer_table "    {\"${entry}\", g_detail_${entry}, sizeof(g_detail_${entry})},\n")
endforeach()
file(CONFIGURE OUTPUT ${INFER_GEN_DIR}/detail_shader_table.inc CONTENT
"// generated by native/infer/CMakeLists.txt
${infer_includes}
struct DetailShader { const char* name; const void* data; size_t size; };
const DetailShader kDetailShaders[] = {
${infer_table}};
" @ONLY)
add_custom_target(nvfx_infer_shaders DEPENDS ${infer_headers})
add_dependencies(nvfx nvfx_infer_shaders)

target_sources(nvfx PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/detail.cpp ${CMAKE_CURRENT_SOURCE_DIR}/nvfx_detail.h)
target_include_directories(nvfx PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${INFER_GEN_DIR}
                           ${NVFX_DML_DIR}/include ${NVFX_ORT_DIR}/build/native/include)

# ---- bench / parity executable (C API only)
add_executable(nvfx_detail_bench ${CMAKE_CURRENT_SOURCE_DIR}/bench.cpp)
target_include_directories(nvfx_detail_bench PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(nvfx_detail_bench PRIVATE nvfx d3d12 dxgi)
target_compile_options(nvfx_detail_bench PRIVATE /W4 /permissive- /EHsc)
target_compile_definitions(nvfx_detail_bench PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
add_custom_command(TARGET nvfx_detail_bench POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different ${NVFX_INFER_DLLS} $<TARGET_FILE_DIR:nvfx_detail_bench>
  VERBATIM)
