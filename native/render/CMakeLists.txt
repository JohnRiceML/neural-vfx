# native/render: volume ray-marcher (C++ + HLSL). OWNER: native render engineer.
# Every compute entry point is compiled with dxc at build time and embedded in nvfx.dll (same pattern as native/sim).
set(RENDER_HLSL_DIR ${CMAKE_CURRENT_SOURCE_DIR}/shaders)
set(RENDER_GEN_DIR ${CMAKE_CURRENT_BINARY_DIR}/gen)
file(MAKE_DIRECTORY ${RENDER_GEN_DIR}/render_shaders)

# file:entry:table_name[:DEFINE=value]
set(RENDER_SHADERS
  prep:r_prep1:prep1 prep:r_prep2:prep2 prep:r_prep3:prep3 prep:r_prep4:prep4 prep:r_prep5:prep5 prep:r_prep6:prep6
  march:r_march:march_c:AUX=0 march:r_march:march_aux:AUX=1
  post:r_bloom1:bloom1 post:r_bloom2:bloom2 post:r_bloom3:bloom3
  post:r_composite:composite_d:DBL=1 post:r_composite:composite_f:DBL=0)

# -Gis: IEEE strictness (no reassociation / contraction, so the AUX and colour-only march variants share the colour
# arithmetic bit for bit). NOT -denorm preserve (unlike native/sim): on this NVIDIA driver any log2 (so any pow)
# compiled with it makes CreateComputePipelineState fail with E_OUTOFMEMORY (minimal repro: one log2 in a 64-thread
# kernel). Denormals cannot matter at 8-bit / fp16 output precision.
set(RENDER_DXC_FLAGS -nologo -T cs_6_6 -HV 2021 -O3 -Gis -WX)
set(RENDER_DEPS ${RENDER_HLSL_DIR}/render_common.hlsli ${CMAKE_CURRENT_SOURCE_DIR}/rslots.h)

set(render_headers "")
set(render_includes "")
set(render_table "")
foreach(item ${RENDER_SHADERS})
  string(REPLACE ":" ";" parts ${item})
  list(GET parts 0 file)
  list(GET parts 1 entry)
  list(GET parts 2 name)
  list(LENGTH parts nparts)
  set(defs "")
  if(nparts GREATER 3)
    list(GET parts 3 d)
    set(defs -D ${d})
  endif()
  set(out ${RENDER_GEN_DIR}/render_shaders/${name}.h)
  add_custom_command(
    OUTPUT ${out}
    COMMAND ${NVFX_DXC} ${RENDER_DXC_FLAGS} -E ${entry} ${defs} -I ${RENDER_HLSL_DIR} -I ${CMAKE_CURRENT_SOURCE_DIR}
            -Vn g_render_${name} -Fh ${out} ${RENDER_HLSL_DIR}/${file}.hlsl
    DEPENDS ${RENDER_HLSL_DIR}/${file}.hlsl ${RENDER_DEPS}
    COMMENT "dxc ${file}.hlsl:${entry} -> ${name}"
    VERBATIM)
  list(APPEND render_headers ${out})
  string(APPEND render_includes "#include \"render_shaders/${name}.h\"\n")
  string(APPEND render_table "    {\"${name}\", g_render_${name}, sizeof(g_render_${name})},\n")
endforeach()
file(CONFIGURE OUTPUT ${RENDER_GEN_DIR}/render_shader_table.inc CONTENT
"// generated by native/render/CMakeLists.txt
${render_includes}
struct RenderShader { const char* name; const void* data; size_t size; };
const RenderShader kRenderShaders[] = {
${render_table}};
" @ONLY)

add_custom_target(nvfx_render_shaders DEPENDS ${render_headers})
add_dependencies(nvfx nvfx_render_shaders)
target_sources(nvfx PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/render.cpp ${CMAKE_CURRENT_SOURCE_DIR}/nvfx_render.h
                            ${CMAKE_CURRENT_SOURCE_DIR}/rslots.h)
target_include_directories(nvfx PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${RENDER_GEN_DIR})

# owner self-check + bench (C API only)
add_executable(nvfx_render_selfcheck ${CMAKE_CURRENT_SOURCE_DIR}/selfcheck.cpp)
target_include_directories(nvfx_render_selfcheck PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(nvfx_render_selfcheck PRIVATE nvfx)
target_compile_options(nvfx_render_selfcheck PRIVATE /W4 /permissive- /EHsc)
target_compile_definitions(nvfx_render_selfcheck PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
