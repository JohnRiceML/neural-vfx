# native/sim: the sim3d solver (C++ + HLSL). OWNER: native solver engineer.
# Every compute entry point is compiled with dxc at build time and embedded in nvfx.dll.
set(SIM_HLSL_DIR ${CMAKE_CURRENT_SOURCE_DIR}/shaders)
set(SIM_GEN_DIR ${CMAKE_CURRENT_BINARY_DIR}/gen)
file(MAKE_DIRECTORY ${SIM_GEN_DIR}/shaders)

# file:entry
set(SIM_SHADERS
  advect:adv_sl advect:adv_mc advect:mass_fin
  forces:sources forces:diffuse forces:curl forces:sep_x forces:sep_y forces:sep_z_store forces:sep_z_apply
  forces:body_forces
  pressure:p_rhs pressure:p_fin_maxu pressure:p_matvec_dot pressure:p_pre pressure:p_resid pressure:p_post_rz
  pressure:p_fin_rz pressure:p_upd_p pressure:p_apply pressure:p_fwd_red pressure:p_smooth_h pressure:p_restrict_red
  pressure:p_prolong_black pressure:p_inv_transform pressure:p_coarse_block pressure:p_smooth pressure:p_prolong
  pressure:p_coarsest pressure:p_cg_update pressure:p_coarse_lds pressure:p_resid_fwd
  setup:s_init setup:s_mask setup:s_probe1 setup:s_galerkin)

# -Gis: IEEE strictness (no fast-math reassociation); -denorm preserve: like CUDA without ftz (Warp default)
set(SIM_DXC_FLAGS -nologo -T cs_6_6 -HV 2021 -O3 -Gis -denorm preserve -WX)

set(sim_headers "")
set(sim_includes "")
set(sim_table "")
foreach(item ${SIM_SHADERS})
  string(REPLACE ":" ";" parts ${item})
  list(GET parts 0 file)
  list(GET parts 1 entry)
  set(out ${SIM_GEN_DIR}/shaders/${entry}.h)
  add_custom_command(
    OUTPUT ${out}
    COMMAND ${NVFX_DXC} ${SIM_DXC_FLAGS} -E ${entry} -I ${SIM_HLSL_DIR} -I ${CMAKE_CURRENT_SOURCE_DIR}
            -Vn g_cs_${entry} -Fh ${out} ${SIM_HLSL_DIR}/${file}.hlsl
    DEPENDS ${SIM_HLSL_DIR}/${file}.hlsl ${SIM_HLSL_DIR}/common.hlsli ${CMAKE_CURRENT_SOURCE_DIR}/slots.h
    COMMENT "dxc ${file}.hlsl:${entry}"
    VERBATIM)
  list(APPEND sim_headers ${out})
  string(APPEND sim_includes "#include \"shaders/${entry}.h\"\n")
  string(APPEND sim_table "    {\"${entry}\", g_cs_${entry}, sizeof(g_cs_${entry})},\n")
endforeach()
file(CONFIGURE OUTPUT ${SIM_GEN_DIR}/shader_table.inc CONTENT
"// generated by native/sim/CMakeLists.txt
${sim_includes}
struct ShaderEntry { const char* name; const void* data; size_t size; };
const ShaderEntry kShaderTable[] = {
${sim_table}};
" @ONLY)

# custom-command outputs only drive targets of this directory: build them through a helper target
add_custom_target(nvfx_sim_shaders DEPENDS ${sim_headers})
add_dependencies(nvfx nvfx_sim_shaders)
target_sources(nvfx PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/solver.cpp ${CMAKE_CURRENT_SOURCE_DIR}/slots.h)
target_include_directories(nvfx PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${SIM_GEN_DIR})

# owner self-check (C API only)
add_executable(nvfx_selfcheck ${CMAKE_CURRENT_SOURCE_DIR}/selfcheck.cpp)
target_link_libraries(nvfx_selfcheck PRIVATE nvfx d3d12)
target_compile_options(nvfx_selfcheck PRIVATE /W4 /permissive- /EHsc)
target_compile_definitions(nvfx_selfcheck PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
